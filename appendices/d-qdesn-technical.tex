\chapter{Posterior computation and supporting evidence for Q--DESN}
\label{app:qdesn-technical}

This appendix records the calculations and empirical checks needed to
interpret Chapter~\ref{ch:research-c}. It distinguishes the single-quantile
posterior from the stacked multiquantile update, states the variational
approximations, and retains sensitivity results that qualify the chapter's
comparisons. Posterior inference is conditional on the selected reservoir
realization and its fixed feature matrix.

\section{Posterior formulation}
\label{app:qdesn-posterior}

\subsection{Conventions and fixed reservoir features}

All inverse-gamma distributions use
\[
p(x\mid a,b)\propto x^{-a-1}\exp(-b/x),
\]
and \(\GIG(\lambda,\chi,\psi)\) has kernel
\[
x^{\lambda-1}\exp\{-(\chi/x+\psi x)/2\}.
\]
The notation
\(\TN(m,V)\) denotes a normal distribution with mean \(m\) and variance
\(V\), truncated to the positive half-line.

For a target level \(p\), the AL constants are
\(A_0=(1-2p)/\{p(1-p)\}\) and \(B_0=2/\{p(1-p)\}\). For exAL, let
\[
g(\gamma)=2\Phi(-|\gamma|)\exp(\gamma^2/2),\qquad
p_\gamma=\ind\{\gamma<0\}+\frac{p-\ind\{\gamma<0\}}{g(\gamma)},
\]
and define
\[
A_\gamma=\frac{1-2p_\gamma}{p_\gamma(1-p_\gamma)},\qquad
B_\gamma=\frac{2}{p_\gamma(1-p_\gamma)},\qquad
D_\gamma=\{\ind(\gamma>0)-p_\gamma\}^{-1}|\gamma|.
\]
At \(\gamma=0\), the exAL hierarchy reduces exactly to AL. Away from zero,
derivatives are evaluated within the appropriate sign branch; two-sided
smoothness at zero is not assumed.

The reservoir matrices, input matrices, reduction maps, leak rates, spectral
radii, and washout are fixed before regression inference. If
\(\vect u_t\) contains response lags and exogenous inputs, the layer states
obey the leaky recursions described in Section~\ref{qdesn2026:sec:model}.
Their fixed regression stack is
\begin{equation}
\tilde{\vect x}_t=\{
\vect h_{t,D}^\top,
k(\tilde{\vect h}_{t,1})^\top,\ldots,
k(\tilde{\vect h}_{t,D-1})^\top,
\vect u_t^\top\}^\top,
\qquad
\vect x_t=(1,\tilde{\vect x}_t^\top)^\top.
\label{qdesn2026:eq:supp_feature_stack}
\end{equation}
Direct inputs are included only where the analysis specifies them. Future
rows that depend on unknown responses are generated by the forecast recursion;
they are not treated as fixed in advance.

\subsection{Single-quantile target}

At a fixed probability level, the exAL mixture representation is
\begin{align}
v_t\mid\sigma&\sim\Exp(\text{rate}=1/\sigma),&
s_t&\sim\TN(0,1),\nonumber\\
y_t\mid\vect\beta,\sigma,\gamma,v_t,s_t,\mat X
&\sim\Normal\left(
\vect x_t^\top\vect\beta+\sigma D_\gamma s_t+A_\gamma v_t,
\ \sigma B_\gamma v_t\right).
\label{qdesn2026:eq:supp_exal_hierarchy}
\end{align}
Integrating out \((v_t,s_t)\) leaves
\(\vect x_t^\top\vect\beta\) as the conditional \(p\)-quantile. The AL
model fixes \(\gamma=0\) and removes \(s_t\).

The Gaussian ridge prior is
\(\vect\beta\sim\Normal(\vect b_\beta,\mat V_\beta)\). The
regularized-horseshoe alternative exempts the intercept and assigns each
slope the product-form prior
\[
p(\beta_j,\lambda_j\mid\tau,\zeta)
\propto
\Normal(\beta_j\mid0,\tau^2\lambda_j^2)
\Normal(\beta_j\mid0,\zeta^2),
\]
with half-Cauchy inverse-gamma augmentations for the local and global scales
and either a fixed slab or \(\zeta^2\sim\IG(a_\zeta,b_\zeta)\). This form
implies conditional precision
\(\tau^{-2}\lambda_j^{-2}+\zeta^{-2}\). The reference global scale
\(\tau_0\) is calibrated from the design dimension, sample size, a residual
information scale, and a prespecified effective model size; it is a prior
hyperparameter, not a fitted global scale.

Let \(f\in\{\mathrm{AL},\mathrm{exAL}\}\), let
\(\vect a_f=\vect v\) for AL and
\(\vect a_f=(\vect v^\top,\vect s^\top)^\top\) for exAL, and let
\(\eta_f\) be empty for AL and equal to \(\gamma\) for exAL. The common
augmented target is
\begin{equation}
\begin{aligned}
&p(\vect y,\vect a_f,\vect\beta,\sigma,\eta_f,\Theta_h\mid\mat X)\\
&\quad\propto
\prod_{t=1}^T
\{p_f(y_t\mid\vect\beta,\sigma,\eta_f,\vect a_{f,t},\vect x_t)
p_f(\vect a_{f,t}\mid\sigma)\}\,
p_h(\vect\beta,\Theta_h)p(\sigma)p_f(\eta_f),
\end{aligned}
\label{qdesn2026:eq:supp-generic-augmented-target}
\end{equation}
where \(h\) identifies ridge or regularized-horseshoe shrinkage. Each
mixture variable and each prior block appears once.

\subsection{Stacked multiquantile target}

For \(0<p_1<\cdots<p_K<1\), write
\(\mu_{k,t}=\beta_{0k}+\tilde{\vect x}_t^\top\vect\beta_k\). The exAL
augmentation is copied across levels, with level-specific
\((\sigma_k,\gamma_k,v_{k,t},s_{k,t})\). After stacking the response \(K\)
times, the augmented composite likelihood is
\begin{equation}
L_{\mathrm{aug}}
=\phi_{KT}\!\left(
\vect y_{\mathrm{st}};
\vect\beta_{0,\mathrm{st}}+
(\mat I_K\otimes\mat Z)\vect\beta_{\mathrm{st}}+\vect d,
\mat\Omega\right),
\label{qdesn2026:eq:supp_joint_stacked_likelihood}
\end{equation}
where \(\mat\Omega\) is diagonal with entries
\(\sigma_kB_kv_{k,t}\). This density evaluates the product of the \(KT\)
quantile-specific Gaussian augmentation factors. The repeated response is an
evaluation argument; the construction is an untempered composite update, not
a normalized \(KT\)-variate sampling law for the observed response.

The reported MQ--DESN fits fix the baseline slope at zero and regularize
\(\Delta_1=\vect\beta_1\) and
\(\Delta_k=\vect\beta_k-\vect\beta_{k-1}\), \(k\ge2\), with separate
regularized-horseshoe blocks. If \(\mat H\vect\beta_{\mathrm{st}}\) stacks
these differences and \(\mat D_\Delta\) contains their conditional prior
variances, then
\begin{equation}
p(\vect\beta_{\mathrm{st}}\mid\mat D_\Delta)
\propto |\mat D_\Delta|^{-1/2}
\exp\{-\tfrac12
(\mat H\vect\beta_{\mathrm{st}})^\top
\mat D_\Delta^{-1}(\mat H\vect\beta_{\mathrm{st}})\}.
\label{qdesn2026:eq:supp_joint_qvp_prior}
\end{equation}
The resulting complete-data posterior multiplies
Eq.~\eqref{qdesn2026:eq:supp_joint_stacked_likelihood}, all latent-variable
densities, level-specific likelihood priors, the ordered-intercept prior, and
the complete difference-scale hierarchies. A random baseline is a possible
extension but is not part of the reported study. Ordered intercepts do not by
themselves prevent crossing when slopes differ.

\section{Posterior computation}
\label{app:qdesn-computation}

\subsection{MCMC blocks}

For exAL, define
\(y_t^*=y_t-\sigma D_\gamma s_t-A_\gamma v_t\) and
\(w_t=(\sigma B_\gamma v_t)^{-1}\). With
\(\mat W=\diag(w_t)\), prior precision \(\mat P\), and prior mean
\(\vect b\), the coefficient update is
\begin{align}
\mat\Sigma_\beta&=(\mat X^\top\mat W\mat X+\mat P)^{-1},\nonumber\\
\vect m_\beta&=\mat\Sigma_\beta
(\mat X^\top\mat W\vect y^*+\mat P\vect b),\qquad
\vect\beta\mid\cdot\sim\Normal(\vect m_\beta,\mat\Sigma_\beta).
\label{qdesn2026:eq:supp_beta_mcmc}
\end{align}
The ridge and regularized-horseshoe cases differ only through \(\mat P\)
and the scale updates. The exAL latent full conditionals are GIG for
\(v_t\) and positive-truncated normal for \(s_t\); conditional on \(\gamma\),
the scale is GIG. The bounded \(\gamma\) coordinate is nonconjugate and is
updated on a logit transformation within its sign branch. Under AL, the
\(s_t\) and \(\gamma\) blocks disappear and \(\sigma\) is inverse-gamma.
The regularized-horseshoe scale blocks have inverse-gamma full conditionals.

For MQ--DESN, let
\(\vect r=\vect y_{\mathrm{st}}-\vect\beta_{0,\mathrm{st}}-\vect d\).
The stacked slope precision and mean are
\begin{align}
\mat K_\beta
&=(\mat I_K\otimes\mat Z)^\top\mat\Omega^{-1}
(\mat I_K\otimes\mat Z)
+\mat H^\top\mat D_\Delta^{-1}\mat H,\nonumber\\
\vect m_\beta
&=\mat K_\beta^{-1}(\mat I_K\otimes\mat Z)^\top
\mat\Omega^{-1}\vect r.
\label{qdesn2026:eq:supp_joint_beta_conditional}
\end{align}
Thus \(\vect\beta_{\mathrm{st}}\mid\cdot\) is Gaussian with covariance
\(\mat K_\beta^{-1}\). A sparse factorization is used rather than a dense
inverse. The remaining latent, likelihood, intercept, and shrinkage blocks
are level-indexed versions of the single-quantile updates.

\begin{algorithmblock}{Q--DESN and MQ--DESN MCMC updates}
\label{qdesn2026:alg:qdesn_mcmc}
Initialize coefficients, likelihood parameters, mixture variables, and
shrinkage scales. At each iteration: update the GIG latent scales; update the
positive-shift variables for exAL; draw the single-level or sparse stacked
Gaussian coefficient block; update the inverse-gamma shrinkage scales; update
\(\sigma\) directly for AL or the transformed scale--shape block for exAL;
and retain draws after burn-in together with chain and crossing diagnostics.
For MQ--DESN, intercepts use their Gaussian conditionals restricted to their
ordered support.
\end{algorithmblock}

\subsection{Variational coordinates and structured scale--shape factor}

The single-level variational family retains a full Gaussian covariance for
the regression coefficients and uses mean-field factors for latent and
shrinkage scales. Conditional conjugacy supplies Gaussian, GIG,
positive-truncated-normal, and inverse-gamma coordinate updates. The earlier
Laplace--delta calculation approximates transformed \((\sigma,\gamma)\) by a
Gaussian at the local mode and uses second-order delta moments. The reported
multiquantile study instead uses a structured
\(q(\gamma)q(\sigma\mid\gamma)\) factor.

For one level, let \(e_t=y_t-\mu_t\), with the intercept held at its current
point value, and denote the required moments by
\(\bar e_t=\E(e_t)\), \(e_t^{(2)}=\E(e_t^2)\),
\(\bar v_t=\E(v_t)\), \(v_t^-=\E(v_t^{-1})\),
\(\bar s_t=\E(s_t)\), and \(s_t^{(2)}=\E(s_t^2)\). Then
\begin{align}
\nu&=-a_\sigma-3T/2,\nonumber\\
\chi_\gamma&=2b_\sigma+2\sum_t\bar v_t+
B_\gamma^{-1}\sum_t
\{e_t^{(2)}v_t^- -2A_\gamma\bar e_t+A_\gamma^2\bar v_t\},\nonumber\\
\psi_\gamma&=D_\gamma^2B_\gamma^{-1}\sum_t s_t^{(2)}v_t^-,\nonumber\\
c_\gamma&=\log\pi_\gamma(\gamma)-\tfrac T2\log B_\gamma+
D_\gamma B_\gamma^{-1}\sum_t\bar s_t(\bar e_tv_t^- -A_\gamma).
\label{qdesn2026:eq:supp_structured_scale_shape_terms}
\end{align}
The coordinate kernel is
\begin{equation}
q^*(\sigma,\gamma)\propto
\exp(c_\gamma)\sigma^{\nu-1}
\exp\{-\tfrac12(\chi_\gamma/\sigma+\psi_\gamma\sigma)\}
\ind\{\sigma>0,L<\gamma<U\}.
\label{qdesn2026:eq:supp_structured_scale_shape_kernel}
\end{equation}
Consequently, \(q(\sigma\mid\gamma)=\GIG(\nu,\chi_\gamma,\psi_\gamma)\).
Its unnormalized integral is
\begin{equation}
Z_\sigma(\gamma)=
\begin{cases}
2(\chi_\gamma/\psi_\gamma)^{\nu/2}
K_\nu(\sqrt{\chi_\gamma\psi_\gamma}),&\psi_\gamma>0,\\
\Gamma(-\nu)(\chi_\gamma/2)^\nu,&\psi_\gamma=0,\ \nu<0.
\end{cases}
\label{qdesn2026:eq:supp_structured_scale_integral}
\end{equation}
Retaining this \(\gamma\)-dependent mass gives
\begin{equation}
q(\gamma)=\frac{\exp(c_\gamma)Z_\sigma(\gamma)}
{(\int_L^0+\int_0^U)\exp(c_u)Z_\sigma(u)\,du}.
\label{qdesn2026:eq:supp_structured_shape_factor}
\end{equation}
One-dimensional quadrature split at zero supplies mixed moments:
\begin{equation}
\E\{h(\sigma,\gamma)\}
=\frac{(\int_L^0+\int_0^U)
\exp(c_\gamma)Z_\sigma(\gamma)
\E\{h(\sigma,\gamma)\mid\gamma\}\,d\gamma}{
(\int_L^0+\int_0^U)\exp(c_\gamma)Z_\sigma(\gamma)\,d\gamma}.
\label{qdesn2026:eq:supp_structured_mixed_moments}
\end{equation}
This dependence matters: replacing these quantities by products of separate
scale and shape expectations defines a different approximation. For the
stacked Gaussian slope factor, adjacent-difference second moments also retain
cross-level covariance through
\[
\E\{(\mat H\vect\beta)\circ(\mat H\vect\beta)\}
=(\mat H\vect m)\circ(\mat H\vect m)
+\diag(\mat H\mat\Sigma\mat H^\top).
\]

\begin{algorithmblock}{Variational coordinate updates}
\label{qdesn2026:alg:qdesn_vb_ld}
Initialize the Gaussian coefficient factor, latent-variable factors,
likelihood scales, and shrinkage factors. Iterate the Gaussian coefficient,
GIG latent-scale, positive-shift, and inverse-gamma shrinkage updates; update
the transformed Laplace--delta block or the structured scale--shape factor;
then recompute the stated objective and all parameter-change diagnostics.
Terminate only after the required consecutive complete iterations satisfy
every tolerance. The Laplace--delta objective contains approximated
expectations and is therefore a numerical objective, not automatically a
rigorous lower bound.
\end{algorithmblock}

\section{Simulation design and sensitivity evidence}
\label{app:qdesn-simulation-support}

For fitted levels \(p_1<\cdots<p_K\), ordered output uses equal-weight
isotonic projection when the analysis specifies a correction:
\[
\vect q_t^*=\argmin_{q_1\le\cdots\le q_K}
\sum_{k=1}^K(\widehat q_{k,t}-q_k)^2.
\]
This projection differs from sorting and from the probability-integration
weights used in scoring. On the common seven-level grid, the reported
finite-grid score is
\begin{equation}
\aCRPS(y;\vect q^*)=2\sum_{k=1}^K\omega_k
\rho_{p_k}(y-q_k^*),
\qquad
\omega_k=\tfrac12(p_{k+1}-p_{k-1})
\label{qdesn2026:eq:supp-acrps}
\end{equation}
with endpoint weights defined by the adjacent half-widths. The weights sum to
\(p_K-p_1=0.90\), so these values are comparable within a study using the
same grid, not with a full-distribution CRPS on a different range.

\subsection{Multiquantile recursive forecast sensitivity}

The multiquantile experiment uses eight mechanisms, seven fitted levels, 500
fitting observations, and 990 forecast pairs. One scenario-specific reservoir
is selected on separate development series and held common across Q--DESN and
MQ--DESN under AL and exAL. MCMC uses five independently initialized chains.
The primary score evaluates coefficient draws at a future feature matrix
averaged over recursive response paths. This conditions on that averaged
matrix and is not generally the quantile function of a complete posterior
predictive distribution.

\input{tables/ch04-qdesn/qdesn_multiquantile_recursive_policy_display.tex}

The averaged-design intervals are narrower in 28 of 32 comparisons, with
median width ratio 0.7364, because the trajectory-specific calculation
propagates an additional source of uncertainty. Imperfect scale--shape mixing
remains a limitation; stable score distributions do not establish mixing of
the original parameter chains. The descriptive MQ--DESN-minus-Q--DESN score
intervals favor MQ--DESN under both likelihoods only for asymmetric tails;
the other fourteen include zero. All scored grids are ordered by the stated
isotonic projection.

\section{PriceFM comparison}
\label{app:qdesn-pricefm-support}

The PriceFM comparison evaluates average quantile loss over 38 regions and
three temporal folds. Reservoir and shrinkage selection uses three expanding
development periods within the first fold's training span; the outer first-fold
validation period chooses AL or exAL. One regional specification is then fixed
for all three test folds. The same calendar periods had appeared in an earlier
historical analysis, so the present rule is prospective only relative to the
fixed comparison, not to previously unseen dates.

Predictors use released load, solar, and wind columns described as day-ahead
forecasts. Their issue-time availability was not independently verified;
therefore, the comparison does not establish operational information parity.
Training-only median and interquartile-range scaling is used. Eleven regions
select AL and 27 select exAL. Mean AQL is 7.217 for Q--DESN and 7.039 for
PriceFM; Q--DESN is lower in 54 of 114 region--fold comparisons and 16 of 38
region means. Its fold-level advantage occurs only in fold 2.

\section{GloFAS reference--discrepancy correction}
\label{app:qdesn-glofas-support}

The streamflow analysis combines USGS reference observations, retrospective
GloFAS, and a 51-member issued GloFAS ensemble. Separate reservoirs represent
the reference and the GloFAS-minus-reference discrepancy. For level \(p_k\),
\begin{align}
q_{k,t}&=\vect x_t^{\mathrm{ref}\top}\vect\beta_k,&
d_{k,t}&=\vect x_t^{\mathrm{disc}\top}\vect\delta_k,&
q_{k,t}^{\mathrm{glo}}&=q_{k,t}+d_{k,t}.
\label{qdesn2026:eq:supp_glofas_quantile_decomposition}
\end{align}
This is a quantile-indexed location adjustment, not a general identity for
quantiles of differences. Future reference and discrepancy features depend
strictly on earlier elements of the latent 28-day reference path. Withheld
USGS observations are reserved for scoring. Forecast-period weather inputs
are realized PRISM and ERA5 quantities.

Historical selection uses rainy-season development and confirmation periods,
four reservoir realizations, and additional deterioration screens. The final
December 2022--January 2023 period is not used for reservoir, prior, stopping,
or family selection.

For source \(c\), the AL augmentation has latent scale
\(v_{i,k}^c\sim\Exp(1/\sigma_{c,k})\) and Gaussian conditional observation
with location \(\vect h_i^{c\top}\vect\theta_k+A_{0,k}v_{i,k}^c\) and
variance \(\sigma_{c,k}B_{0,k}v_{i,k}^c\). Historical rows have weight one;
each issued member has weight \(1/51\), so the members at a horizon have total
weight one. The fitted target is
\begin{equation}
L_{\mathrm{aug}}(\Theta)\propto
\prod_{k,c}\prod_{i\in\mathcal I_c}
\{p(z_i^c\mid v_{i,k}^c,\Theta_k)
p(v_{i,k}^c\mid\sigma_{c,k})\}^{w_i}.
\label{qdesn2026:eq:supp_glofas_weighted_augmented_likelihood}
\end{equation}
This is a complete-data fractional update. Because the power is applied before
integrating out the mixture variables, it is not claimed to be an exact
fractional marginal AL posterior.

The issued-ensemble quantile fits use a Gaussian future-path prior estimated
from 500 recursive paths under the combined historical Gaussian model. Neither
issued members nor withheld USGS observations enter that prior. At each
coordinate update, reservoir features are linearized around the current path
mode. If \(\widetilde\ell_F\) is the resulting surrogate, then
\begin{equation}
\mat K_F=-\nabla^2\widetilde\ell_F(\widehat{\vect y}_F),
\qquad
q_F(\vect y_F)\approx
\Normal(\widehat{\vect y}_F,\mat K_F^{-1}).
\label{qdesn2026:eq:supp_glofas_future_path_factor}
\end{equation}
For design Jacobian \(\mat J_i\) and \(\mat V_F=\mat K_F^{-1}\), the
remaining updates use
\begin{equation}
\E(\vect h_i^c\vect h_i^{c\top})\approx
\vect a_i^c\vect a_i^{c\top}+\mat J_i\mat V_F\mat J_i^\top.
\label{qdesn2026:eq:supp_glofas_future_path_delta}
\end{equation}
Repeated first-order updates give a local Gaussian--delta approximation, not
an exact Laplace approximation to the nonlinear posterior.

Both MQ--DESN fits meet their stated inner and outer stopping criteria. Over
the 28 held-out horizons, Q--DESN under AL has the lowest descriptive grid
score, but its nominal 90\% interval covers 0.750 of the outcomes. MQ--DESN
under AL covers 0.786. At \(p=0.95\), the empirical frequency for MQ--DESN
under exAL is 0.286, compared with 0.750 for Q--DESN under AL. Thus the score
ranking does not remove the pronounced upper-tail calibration failure.

\section{Exact Gaussian baseline and initialization}
\label{app:qdesn-gaussian-baseline}

Conditional on the fixed design, the Gaussian DESN is a mean-regression
baseline rather than a quantile model:
\begin{align}
\vect y\mid\vect\beta,\omega^2,\mat X
&\sim\Normal_T(\mat X\vect\beta,\omega^2\mat I_T),\nonumber\\
\vect\beta\mid\omega^2
&\sim\Normal_d(\vect b,\omega^2\mat P^{-1}),\qquad
\omega^2\sim\IG(a_\omega,b_\omega).
\label{qdesn2026:gauss:eq:scaled_prior}
\end{align}
The scaling of the coefficient prior by \(\omega^2\) is the conjugacy
condition. Define
\begin{align}
\mat P_n&=\mat P+\mat X^\top\mat X,&
\vect m_n&=\mat P_n^{-1}(\mat P\vect b+\mat X^\top\vect y),\nonumber\\
a_n&=a_\omega+T/2,&
B_n&=b_\omega+\tfrac12\{
\vect y^\top\vect y+\vect b^\top\mat P\vect b
-\vect m_n^\top\mat P_n\vect m_n\}.
\label{qdesn2026:gauss:eq:Bn}
\end{align}
The exact posterior is
\begin{align}
\omega^2\mid\vect y,\mat X&\sim\IG(a_n,B_n),\nonumber\\
\vect\beta\mid\omega^2,\vect y,\mat X
&\sim\Normal_d(\vect m_n,\omega^2\mat P_n^{-1}),\nonumber\\
\vect\beta\mid\vect y,\mat X
&\sim\StudentT_{2a_n}(\vect m_n,(B_n/a_n)\mat P_n^{-1}).
\label{qdesn2026:gauss:eq:beta_student_posterior}
\end{align}
For fixed future row \(\vect x_*\),
\begin{equation}
y_*\mid\vect y,\mat X,\vect x_*
\sim\StudentT_{2a_n}\left(
\vect x_*^\top\vect m_n,
\frac{B_n}{a_n}\{1+\vect x_*^\top\mat P_n^{-1}\vect x_*\}
\right).
\label{qdesn2026:gauss:eq:predictive_univariate}
\end{equation}
For recursively generated future features, prediction propagates posterior
draws through the reservoir rather than treating the entire future design as
fixed.

\begin{algorithmblock}{Direct Gaussian posterior sampler}
\label{qdesn2026:gauss:alg:direct_sampler}
Compute \((\mat P_n,\vect m_n,a_n,B_n)\). Independently draw
\(\omega^{2(m)}\sim\IG(a_n,B_n)\), then draw
\(\vect\beta^{(m)}\sim\Normal(\vect m_n,
\omega^{2(m)}\mat P_n^{-1})\). For a fixed future design, draw responses from
the Gaussian observation law or use the analytic Student distribution. No
Markov chain or variational approximation is required.
\end{algorithmblock}

For a target level \(p\), initialize the quantile readout by shifting the
Gaussian intercept. If \(r_t^N=y_t-\vect x_t^\top\vect m_n\) and
\(c_p\) is their empirical \(p\)-quantile, set
\begin{align}
\beta_0^{\mathrm{init}}(p)&=m_{n,0}+c_p,&
\beta_j^{\mathrm{init}}(p)&=m_{n,j},\quad j\ge1.
\label{qdesn2026:gauss:eq:init_beta0}
\end{align}
The likelihood scale begins at the mean check loss plus a positive numerical
floor; exAL begins at \(\gamma=0\) (or a nearby interior value), with
\(v_t=\sigma\) and \(s_t=\sqrt{2/\pi}\). The Gaussian covariance
\((B_n/(a_n-1))\mat P_n^{-1}\) can initialize the variational coefficient
factor, but it is not an uncertainty statement for the quantile posterior.
