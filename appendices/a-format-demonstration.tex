\section{Equation and proof}
For arbitrary real numbers $x_1,\ldots,x_n$ and $n\geq1$, define
\begin{equation}
 \bar{x}=\frac{1}{n}\sum_{i=1}^{n}x_i,
 \qquad S(a)=\sum_{i=1}^{n}(x_i-a)^2.
 \label{eq:demo-squares}
\end{equation}
\begin{samepage}
\begin{theorem}[Elementary demonstration]
The function $S$ in Equation~\eqref{eq:demo-squares} has the unique minimizer
$a=\bar{x}$.
\end{theorem}
\begin{proof}
Expand $x_i-a=(x_i-\bar{x})+(\bar{x}-a)$ and use
$\sum_i(x_i-\bar{x})=0$ to obtain
$S(a)=S(\bar{x})+n(a-\bar{x})^2$. Since $n>0$, equality with
$S(\bar{x})$ holds exactly when $a=\bar{x}$.
\end{proof}
\end{samepage}

\section{Algorithm, figure and table}
\begin{algorithm}[Elementary summation demonstration]
Input $x_1,\ldots,x_n$ with $n\geq1$. Set $s=0$. For $i=1,\ldots,n$,
replace $s$ by $s+x_i$. Return $s/n$. This is a typesetting example.
\end{algorithm}
